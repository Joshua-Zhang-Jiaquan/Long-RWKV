from __future__ import annotations

import hashlib
import json
import shutil
import subprocess
import sys
import tempfile
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path


repo = Path(sys.argv[1]).absolute()
evidence = Path(sys.argv[2]).absolute()
sys.path.insert(0, str(repo))

from scale.experiments.nonlatent_iclr import service
from scale.experiments.nonlatent_iclr.service import AuditPaths


def digest(path: Path) -> str:
    value = hashlib.sha256()
    with path.open("rb") as handle:
        while chunk := handle.read(1024 * 1024):
            value.update(chunk)
    return value.hexdigest()


commands: list[dict[str, object]] = []


def cli(arguments: list[str]) -> dict[str, object]:
    completed = subprocess.run(
        [sys.executable, "-m", "scale.experiments.nonlatent_iclr.cli", *arguments],
        cwd=repo,
        capture_output=True,
        text=True,
        check=False,
    )
    try:
        parsed: object = json.loads(completed.stdout)
    except json.JSONDecodeError:
        parsed = None
    result: dict[str, object] = {
        "argv": arguments,
        "exit_code": completed.returncode,
        "stdout": completed.stdout,
        "stderr": completed.stderr,
        "parsed": parsed,
    }
    commands.append(result)
    return result


def status(result: dict[str, object]) -> str | None:
    parsed = result["parsed"]
    return parsed.get("status") if isinstance(parsed, dict) else None


def fixture(base: Path, name: str, metadata: bool = False) -> AuditPaths:
    fixture_root = base / name
    fixture_repo = fixture_root / "repo"
    snapshot = fixture_repo / "DAN/v7_arch_round"
    for _, relative, _ in service.SOURCES:
        source = repo / "DAN/v7_arch_round" / relative
        target = snapshot / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(source, target)
    external = fixture_root / "external"
    external.mkdir(parents=True)
    if metadata:
        for relative in service.METADATA_PATHS:
            target = external / relative
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_text('{"step": 4750, "tokens_seen": 4980736000.0}\n', encoding="utf-8")
    return AuditPaths.from_roots(
        fixture_repo,
        fixture_root / "evidence",
        external_root=external,
        live_root=fixture_root / "live",
        staged_root=fixture_root / "staged",
    )


actual_common = [
    "--task", "1",
    "--repo-root", str(repo),
    "--evidence-root", str(evidence),
]
ledger_path = repo / "DAN/nonlatent_iclr/evidence_ledger.json"
scope_paths = [repo / "DAN/v7_arch_round" / relative for _, relative, _ in service.SOURCES]
scope_paths.extend((
    repo / ".omo/plans/nonlatent-rnn-dlm-iclr-research.md",
    repo / "DAN/v7_arch_round/scripts/run_prune_and_launch_ext.sh",
))
scope_before = {str(path): digest(path) if path.is_file() else None for path in scope_paths}
ledger_before = digest(ledger_path)
categories: dict[str, dict[str, object]] = {}
work_path: Path | None = None

with tempfile.TemporaryDirectory(prefix=".manual-qa-work-", dir=evidence) as temporary:
    work = Path(temporary)
    work_path = work

    first = cli(["verify", "--case", "happy", *actual_common])
    second = cli(["verify", "--case", "happy", *actual_common])
    analyzed = cli(["analyze", *actual_common])
    categories["normal_path_repeatability"] = {
        "passed": all(result["exit_code"] == 0 and status(result) == "AUDIT_COMPLETE" for result in (first, second, analyzed)),
        "statuses": [status(first), status(second), status(analyzed)],
    }

    invalid_cli = cli(["prepare", "--task", "not-an-integer", "--repo-root", str(repo)])
    malformed = fixture(work, "malformed")
    malformed_prepare = service.prepare_task_one(malformed)
    malformed_data = json.loads(malformed.ledger_path.read_text(encoding="utf-8"))
    malformed_data["artifacts"].append(dict(malformed_data["artifacts"][0]))
    malformed_payload = (json.dumps(malformed_data, indent=2, sort_keys=True) + "\n").encode()
    malformed.ledger_path.write_bytes(malformed_payload)
    malformed_receipt = malformed.receipt_path(malformed_data["attempt"])
    malformed_receipt.write_text(json.dumps({
        "schema_version": 1,
        "attempt": malformed_data["attempt"],
        "ledger_sha256": hashlib.sha256(malformed_payload).hexdigest(),
        "status": "AUDIT_COMPLETE",
    }, indent=2, sort_keys=True) + "\n", encoding="utf-8")
    malformed_verify = service.verify_task_one(malformed)
    categories["invalid_input_and_schema"] = {
        "passed": (
            invalid_cli["exit_code"] == 2
            and status(invalid_cli) == "INVALID_ARGUMENT"
            and malformed_prepare.status == "AUDIT_COMPLETE"
            and malformed_verify.status == "MALFORMED"
        ),
        "cli_status": status(invalid_cli),
        "schema_status": malformed_verify.status,
    }

    corruption = cli(["verify", "--case", "failure", *actual_common])
    categories["source_corruption"] = {
        "passed": corruption["exit_code"] == 0 and status(corruption) == "EXPECTED_FAILURE_CONFIRMED",
        "status": status(corruption),
    }

    stale = fixture(work, "stale-inventory")
    stale_prepare = service.prepare_task_one(stale)
    appeared = stale.external_root / "m2_baseline_triangle/ability_curve_m4loop.csv"
    appeared.parent.mkdir(parents=True)
    appeared.write_text("4750,4980736000.0,1.222e+20,47.00\n", encoding="utf-8")
    stale_verify = service.verify_task_one(stale)
    categories["stale_inventory"] = {
        "passed": stale_prepare.status == "AUDIT_COMPLETE" and stale_verify.status == "STALE",
        "status": stale_verify.status,
        "detail": stale_verify.detail,
    }

    drift = fixture(work, "external-drift", metadata=True)
    drift_prepare = service.prepare_task_one(drift)
    drift_target = drift.external_root / service.METADATA_PATHS[0]
    drift_target.write_text('{"step": 4751, "tokens_seen": 4981784576.0}\n', encoding="utf-8")
    drift_verify = service.verify_task_one(drift)
    categories["external_source_drift"] = {
        "passed": drift_prepare.status == "AUDIT_COMPLETE" and drift_verify.status == "STALE",
        "status": drift_verify.status,
        "detail": drift_verify.detail,
    }

    real_write_once = service.write_once
    ledger_fault = fixture(work, "ledger-fault")

    def fail_ledger(root: Path, relative: Path, data: bytes) -> bool:
        if relative.name == "evidence_ledger.json":
            raise OSError(5, "manual ledger interruption")
        return real_write_once(root, relative, data)

    try:
        service.write_once = fail_ledger
        ledger_interrupted = service.prepare_task_one(ledger_fault)
    finally:
        service.write_once = real_write_once
    ledger_failure_left_no_receipt = not tuple(ledger_fault.evidence_root.rglob("prepare_receipt.json"))
    ledger_recovered = service.prepare_task_one(ledger_fault)

    receipt_fault = fixture(work, "receipt-fault")

    def fail_receipt(root: Path, relative: Path, data: bytes) -> bool:
        if relative.name == "prepare_receipt.json":
            raise OSError(5, "manual receipt interruption")
        return real_write_once(root, relative, data)

    try:
        service.write_once = fail_receipt
        receipt_interrupted = service.prepare_task_one(receipt_fault)
        receipt_pending = service.verify_task_one(receipt_fault)
    finally:
        service.write_once = real_write_once
    receipt_recovered = service.prepare_task_one(receipt_fault)
    categories["interrupted_publication"] = {
        "passed": (
            ledger_interrupted.status == "PUBLICATION_FAILED"
            and ledger_failure_left_no_receipt
            and ledger_recovered.status == "AUDIT_COMPLETE"
            and receipt_interrupted.status == "PUBLICATION_PENDING"
            and receipt_pending.status == "PUBLICATION_PENDING"
            and receipt_recovered.status == "AUDIT_COMPLETE"
        ),
        "statuses": [
            ledger_interrupted.status,
            ledger_recovered.status,
            receipt_interrupted.status,
            receipt_pending.status,
            receipt_recovered.status,
        ],
    }

    competing = fixture(work, "competing")
    with ThreadPoolExecutor(max_workers=4) as pool:
        competing_results = tuple(pool.map(lambda _: service.prepare_task_one(competing), range(4)))
    competing_payload = competing.ledger_path.read_bytes()
    competing_ledger = json.loads(competing_payload)
    competing_receipt = json.loads(competing.receipt_path(competing_ledger["attempt"]).read_text(encoding="utf-8"))
    repeat_digest = digest(competing.ledger_path)
    repeated = service.prepare_task_one(competing)
    categories["repeated_and_competing_calls"] = {
        "passed": (
            all(result.status in {"AUDIT_COMPLETE", "PUBLICATION_EXISTS"} for result in competing_results)
            and service.verify_task_one(competing).status == "AUDIT_COMPLETE"
            and competing_receipt["ledger_sha256"] == hashlib.sha256(competing_payload).hexdigest()
            and repeated.status == "PUBLICATION_EXISTS"
            and digest(competing.ledger_path) == repeat_digest
        ),
        "competing_statuses": [result.status for result in competing_results],
        "repeat_status": repeated.status,
    }

    contained = fixture(work, "containment")
    contained_prepare = service.prepare_task_one(contained)
    outside = work / "outside.md"
    contained_common = [
        "--task", "1",
        "--repo-root", str(contained.repo_root),
        "--evidence-root", str(contained.evidence_root),
        "--external-root", str(contained.external_root),
        "--live-root", str(contained.live_root),
        "--staged-root", str(contained.staged_root),
    ]
    containment = cli(["render", "--out", str(outside), *contained_common])
    categories["output_containment"] = {
        "passed": (
            contained_prepare.status == "AUDIT_COMPLETE"
            and containment["exit_code"] == 2
            and status(containment) == "UNSAFE_DESTINATION"
            and not outside.exists()
        ),
        "status": status(containment),
        "outside_exists": outside.exists(),
    }

    repeated_actual = cli(["prepare", *actual_common])
    repeated_category = categories["repeated_and_competing_calls"]
    repeated_category["passed"] = bool(repeated_category["passed"]) and (
        repeated_actual["exit_code"] == 2
        and status(repeated_actual) == "PUBLICATION_EXISTS"
        and digest(ledger_path) == ledger_before
    )
    repeated_category["actual_repeat_status"] = status(repeated_actual)
    repeated_category["generated_ledger_unchanged"] = digest(ledger_path) == ledger_before

cleanup_complete = work_path is not None and not work_path.exists()
scope_after = {str(path): digest(path) if path.is_file() else None for path in scope_paths}
categories["scope_preservation"] = {
    "passed": scope_before == scope_after and cleanup_complete and digest(ledger_path) == ledger_before,
    "snapshot_plan_unrelated_hashes_unchanged": scope_before == scope_after,
    "temporary_fixture_removed": cleanup_complete,
    "generated_ledger_unchanged": digest(ledger_path) == ledger_before,
    "scheduler_or_gpu_commands_run": False,
}

summary = {
    "category_count": len(categories),
    "normal_path_runs": 2,
    "overall_passed": len(categories) == 9 and all(bool(item["passed"]) for item in categories.values()),
    "categories": categories,
}
(evidence / "manual-qa-commands.json").write_text(json.dumps(commands, indent=2, sort_keys=True) + "\n", encoding="utf-8")
(evidence / "manual-qa-summary.json").write_text(json.dumps(summary, indent=2, sort_keys=True) + "\n", encoding="utf-8")
print(json.dumps(summary, indent=2, sort_keys=True))
raise SystemExit(0 if summary["overall_passed"] else 1)
